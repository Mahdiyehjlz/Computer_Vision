# Extract classes
classes = set()
with open(annotations_file, 'r') as f:
    lines = f.readlines()
    for line in lines:
        parts = line.strip().split()
        if len(parts) < 6: continue
        classes.add(parts[-1])
classes = sorted(classes)
class_to_id = {cls: idx for idx, cls in enumerate(classes)}
# Group annotations by image
image_data = {}
for line in lines:
    parts = line.strip().split()
    img_name = parts[0]
    bbox = list(map(int, parts[1:5]))
    cls_name = parts[5]
    if img_name not in image_data:
        image_data[img_name] = []
    image_data[img_name].append((bbox, cls_name))
def process_images(images_dir, split):
    image_extensions = ['.jpg', '.jpeg', '.png']

    for img_file in os.listdir(images_dir):
        img_base = os.path.splitext(img_file)[0]

        # Skip if we don't have annotations for this image
        if img_base not in image_data:
            continue

        img_path = os.path.join(images_dir, img_file)

        # Copy image to YOLO dataset (only if doesn't exist)
        dest_img_path = os.path.join(base_dir, "images", split, img_file)

        if not os.path.exists(dest_img_path):  # Only copy if image doesn't exist
            shutil.copy(img_path, dest_img_path)
        else:
            print(f"Skipping existing image: {dest_img_path}")

        # Get image dimensions (always read from source)
        with Image.open(img_path) as img:
            width, height = img.size

        # Write labels (only if doesn't exist)
        label_file = os.path.join(base_dir, "labels", split, f"{img_base}.txt")

        if not os.path.exists(label_file):  # Only create if label doesn't exist
            with open(label_file, 'w') as lf:
                for (x_min, y_min, x_max, y_max), cls_name in image_data[img_base]:
                    # Convert to YOLO format (normalized cx, cy, w, h)
                    cx = (x_min + x_max) / (2 * width)
                    cy = (y_min + y_max) / (2 * height)
                    w = (x_max - x_min) / width
                    h = (y_max - y_min) / height
                    cls_id = class_to_id[cls_name]
                    lf.write(f"{cls_id} {cx:.6f} {cy:.6f} {w:.6f} {h:.6f}\n")
        else:
            print(f"Skipping existing label: {label_file}")

# Process train and validation images
process_images(train_images_dir, "train")
process_images(val_images_dir, "val")
